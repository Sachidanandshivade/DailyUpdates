if(!stack.isEmpty() && stack.peek() == s.charAt(i)) {
stack.pop();
}else{
stack.push(s.charAt(i));
}
}
StringBuilder sb = new StringBuilder();
for(char ch:stack){
sb.append(ch);
}
if(sb.length() == 0) {
return "Empty";
}
return sb.toString();
